cmake_minimum_required(VERSION 3.20)
project(SplatoonBootstrap LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(spl_bootstrap_core STATIC src/sha256.cpp src/install.cpp)
target_include_directories(spl_bootstrap_core PUBLIC src)
if(WIN32)
    target_link_libraries(spl_bootstrap_core PUBLIC ole32 shell32 uuid)
endif()
add_executable(SplatoonSetup src/main.cpp src/gui_session.cpp)
target_link_libraries(SplatoonSetup PRIVATE spl_bootstrap_core Threads::Threads)
if(WIN32)
    set_target_properties(SplatoonSetup PROPERTIES WIN32_EXECUTABLE TRUE)
    target_sources(SplatoonSetup PRIVATE src/gui_windows.cpp)
    target_link_libraries(SplatoonSetup PRIVATE comctl32 comdlg32 gdi32 user32)
else()
    target_sources(SplatoonSetup PRIVATE src/gui_linux.cpp)
    target_link_libraries(SplatoonSetup PRIVATE ${CMAKE_DL_LIBS})
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(bootstrap_tests tests/bootstrap_tests.cpp)
    target_link_libraries(bootstrap_tests PRIVATE spl_bootstrap_core)
    add_test(NAME bootstrap_tests COMMAND bootstrap_tests)
    add_executable(session_tests tests/session_tests.cpp src/gui_session.cpp)
    target_include_directories(session_tests PRIVATE src)
    target_link_libraries(session_tests PRIVATE Threads::Threads)
    add_dependencies(session_tests SplatoonSetup)
    add_test(NAME session_tests COMMAND session_tests $<TARGET_FILE:SplatoonSetup>)
    if(WIN32)
        add_executable(windows_log_control_tests tests/windows_log_control_tests.cpp)
        target_link_libraries(windows_log_control_tests PRIVATE user32)
        add_test(NAME windows_log_control_tests COMMAND windows_log_control_tests)
    endif()
endif()
